cmake_minimum_required(VERSION 3.20)
project(Meridian VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(Meridian WIN32 src/main.cpp src/app.rc)
target_include_directories(Meridian PRIVATE third_party)
target_compile_definitions(Meridian PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(Meridian PRIVATE gdiplus comctl32 dwmapi shell32 ole32)
if(MINGW)
  target_compile_options(Meridian PRIVATE -Wall -Wextra -Wpedantic -ffile-prefix-map=${CMAKE_SOURCE_DIR}=. -finput-charset=UTF-8)
  target_link_options(Meridian PRIVATE -municode -static -static-libgcc -static-libstdc++ -s -Wl,--no-insert-timestamp)
elseif(MSVC)
  target_compile_options(Meridian PRIVATE /W4 /utf-8)
endif()
